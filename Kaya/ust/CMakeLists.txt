cmake_minimum_required(VERSION 3.16)
project(ust VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(CURL REQUIRED)
add_executable(ust src/main.cpp)
target_link_libraries(ust PRIVATE CURL::libcurl)
if(MSVC)
  target_compile_options(ust PRIVATE /W4)
else()
  target_compile_options(ust PRIVATE -Wall -Wextra -Wpedantic)
endif()
